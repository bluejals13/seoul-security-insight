# 2. 프로젝트 Python 가상환경 생성

```bash

```


## 버전 확인

```bash

# 1. 현재 Git 상태
git --version
git status
git remote -v

# 2. Python 위치 확인
python --version
which python
python -c "import sys; print(sys.executable)"

# 3. uv 가상환경 생성
uv --version

# 4. 활성화
source .venv/Scripts/activate

# 5. Node.js / npm / npx 확인
node --version
npm --version
npx --version

# 6. Streamlit
streamlit --version

# 7. Python 주요 패키지
python -c "import pandas, numpy, requests, bs4, selenium, streamlit, plotly, dotenv; print('Python packages: OK')"

python -c "import selenium; print(selenium.__version__)"

# 9. Codex CLI
codex --version

# 10. Context7
npx ctx7 --version


# 11. 안티그래비티
```
